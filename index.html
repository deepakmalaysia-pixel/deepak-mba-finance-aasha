<section class="hero">
<p class="eyebrow">A Skills Roadmap for MBA Finance Graduates, India</p>
<h1>An MBA gets you<br>the <em>interview</em>.<br>These get you the <em>offer</em>.</h1>
<p class="lede">Most MBA Finance graduates leave campus knowing the theory of markets, accounting, and valuation — but not the tools recruiters actually test for: a live Excel model, a Tally-ready GST return, a Power BI dashboard, a NISM certificate, a case cracked in fifteen minutes.</p>
<p class="lede">This roadmap is organised the way the finance job market actually is: by <strong>role</strong>, not by subject. Pick the ticker that matches where you want to sit — trading floor, wealth desk, analytics team, audit firm, consulting bay, or corporate finance office — and follow it one quarter at a time.</p>

(function(){
var chips = document.querySelectorAll('.chip');
var tracks = document.querySelectorAll('.track');
var active = new Set();

chips.forEach(function(chip){
chip.addEventListener('click', function(){
var tag = chip.getAttribute('data-tag');
if(active.has(tag)){
active.delete(tag);
chip.classList.remove('active');
} else {
active.add(tag);
chip.classList.add('active');
}
applyFilter();
});
});

function applyFilter(){
if(active.size === 0){
tracks.forEach(function(t){ t.removeAttribute('data-dim'); });
return;
}
tracks.forEach(function(t){
var tags = (t.getAttribute('data-tags') || '').split(' ');
var match = tags.some(function(tag){ return active.has(tag); });
if(match){ t.removeAttribute('data-dim'); }
else { t.setAttribute('data-dim', 'true'); }
});
}
})();
